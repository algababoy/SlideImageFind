/**
 * Enhanced Class Slides Generator with Sections Support
 * - Per-slide multi sections (multiple text boxes per slide)
 * - Each section has: xStart/xEnd (as % width), font family, color, size, alignment, and content
 * - Dual-handle width control implemented with two range inputs (xStart, xEnd) + live preview bar
 * - Sections are included in JSON export/import and HTML export
 * - Backward compatible: legacy title/content still render if no sections present
 */

type SlideType = "title" | "content" | "image";
type TextAlign = "left" | "center" | "right";
type OverlayType = "none" | "horizontal" | "sides";
type FontFamily = "system" | "serif" | "mono" | "display" | "hand";
type TransitionType = "none" | "fade" | "slide" | "zoom" | "bounce" | "flip" | "rotate";

interface SlideSection {
  id: string;
  heading?: string;
  text: string;
  xStart: number; // 0..100 (% of slide width)
  xEnd: number;   // 0..100 (% of slide width) must be > xStart
  yStart: number; // 0..100 (% of slide height)
  yEnd: number;   // 0..100 (% of slide height) must be > yStart
  fontFamily: FontFamily;
  fontSize: number; // rem in editor/preview; px in export
  color: string; // hex
  align: TextAlign;
  transition?: TransitionType;
  transitionDelay?: number; // seconds
}

interface Slide {
  id: string;
  title: string;
  content: string; // legacy single block
  type: SlideType;
  imageUrl?: string;
  imageSource?: ImageSource; // For MLA citations
  backgroundColor?: string;
  textColor?: string; // legacy color for title/content
  overlayType?: OverlayType;
  overlayColor?: string;
  overlayOpacity?: number; // 0..1
  overlayTransition?: TransitionType;
  textAlign?: TextAlign;   // legacy
  titleSize?: number; // rem
  bodySize?: number;  // rem
  sections?: SlideSection[]; // NEW
}

interface UnsplashImage {
  id: string;
  urls: { small: string; regular: string; full: string };
  alt_description: string;
  user: { name: string };
}

interface WikiImage {
  id: string;
  thumb: string;
  full: string;
  title: string;
  author?: string;
  license?: string;
}

interface ImageSource {
  type: 'unsplash' | 'wikimedia' | 'upload';
  url: string;
  title?: string;
  author?: string;
  license?: string;
  accessDate: string;
}

type Provider = "wikimedia" | "unsplash";

export default function ClassSlides() {
  const [presentationTitle, setPresentationTitle] = useState("Présentation de Classe");
  const [slides, setSlides] = useState<Slide[]>(DEFAULT_SLIDES);
  const [currentSlideIndex, setCurrentSlideIndex] = useState(0);
  const [currentSectionIndex, setCurrentSectionIndex] = useState(0);
  const [isPreviewMode, setIsPreviewMode] = useState(false);
  const [showImageSearch, setShowImageSearch] = useState(false);
  const [searchQuery, setSearchQuery] = useState("");
  const [provider, setProvider] = useState<Provider>("wikimedia");
  const [unsplashKey, setUnsplashKey] = useState<string>("");
  const [searching, setSearching] = useState(false);
  const [unsplashResults, setUnsplashResults] = useState<UnsplashImage[]>([]);
  const [wikiResults, setWikiResults] = useState<WikiImage[]>([]);
  const [directImageUrl, setDirectImageUrl] = useState<string>("");
  const [loadingDirectImage, setLoadingDirectImage] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);
  const { toast } = useToast();

  const currentSlide = slides[currentSlideIndex];

  // Utility functions
  const clamp = (n: number, min: number, max: number) => Math.max(min, Math.min(max, n));
  
  const hexWithOpacity = (hex: string, opacity: number) => {
    const clean = hex.replace("#", "");
    const a = clamp(Math.round(clamp(opacity, 0, 1) * 255), 0, 255)
      .toString(16)
      .padStart(2, "0");
    return `#${clean}${a}`;
  };
  
  const luminance = (hex: string) => {
    const clean = hex.replace("#", "");
    const r = parseInt(clean.substring(0, 2), 16) / 255;
    const g = parseInt(clean.substring(2, 4), 16) / 255;
    const b = parseInt(clean.substring(4, 6), 16) / 255;
    const [R, G, B] = [r, g, b].map((v) => (v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4)));
    return 0.2126 * R + 0.7152 * G + 0.0722 * B;
  };
  
  const contrastRatio = (fg: string, bg: string) => {
    const L1 = luminance(fg) + 0.05;
    const L2 = luminance(bg) + 0.05;
    return L1 > L2 ? L1 / L2 : L2 / L1;
  };
  
  const contrastWarning = useMemo(() => {
    const bg = currentSlide?.backgroundColor || "#3B82F6";
    const fg = currentSlide?.textColor || "#ffffff";
    const ratio = contrastRatio(fg, bg);
    return { ratio, warn: ratio < 4.5 };
  }, [currentSlide?.backgroundColor, currentSlide?.textColor]);

  // CRUD operations for slides
  const addSlide = () => {
    setSlides((prev) => {
      const newSlide: Slide = {
        id: Date.now().toString(),
        title: "Nouvelle Diapositive",
        content: "",
        type: "content",
        backgroundColor: "#6366F1",
        textColor: "#ffffff",
        overlayType: "none",
        overlayColor: "#000000",
        overlayOpacity: 0.5,
        overlayTransition: "fade",
        textAlign: "center",
        titleSize: 2.2,
        bodySize: 1.1,
        sections: [
          {
            id: `sec-${Date.now()}`,
            heading: "Texte",
            text: "Votre texte ici...",
            xStart: 15,
            xEnd: 85,
            yStart: 25,
            yEnd: 75,
            fontFamily: "system",
            fontSize: 1.2,
            color: "#ffffff",
            align: "left",
            transition: "fade",
            transitionDelay: 0.5,
          },
        ],
      };
      const next = [...prev, newSlide];
      setCurrentSlideIndex(next.length - 1);
      return next;
    });
  };

  const deleteSlide = (slideId: string) => {
    setSlides((prev) => {
      if (prev.length <= 1) {
        toast({ title: "Impossible de supprimer", description: "Il faut au moins une diapositive.", variant: "destructive" });
        return prev;
      }
      const idx = prev.findIndex((s) => s.id === slideId);
      const next = prev.filter((s) => s.id !== slideId);
      setCurrentSlideIndex((i) => {
        if (idx === -1) return Math.min(i, next.length - 1);
        if (i > idx) return i - 1;
        if (i === idx) return Math.min(i, next.length - 1);
        return i;
      });
      return next;
    });
  };

  const moveSlide = (slideId: string, direction: "up" | "down") => {
    setSlides((prev) => {
      const i = prev.findIndex((s) => s.id === slideId);
      if (i === -1) return prev;
      const j = direction === "up" ? i - 1 : i + 1;
      if (j < 0 || j >= prev.length) return prev;
      const next = [...prev];
      [next[i], next[j]] = [next[j], next[i]];
      setCurrentSlideIndex((idx) => (idx === i ? j : idx === j ? i : idx));
      return next;
    });
  };

  const updateSlide = (slideId: string, updates: Partial<Slide>) => {
    setSlides((prev) => prev.map((s) => (s.id === slideId ? { ...s, ...updates } : s)));
  };

  // CRUD operations for sections
  const addSection = (slideId: string) => {
    setSlides((prev) =>
      prev.map((s) =>
        s.id === slideId
          ? {
              ...s,
              sections: [
                ...(s.sections || []),
                {
                  id: `sec-${Date.now()}`,
                  heading: "Section",
                  text: "Texte...",
                  xStart: 10,
                  xEnd: 90,
                  yStart: 20,
                  yEnd: 80,
                  fontFamily: "system",
                  fontSize: 1.1,
                  color: "#ffffff",
                  align: "left",
                  transition: "fade",
                  transitionDelay: 0.5,
                },
              ],
            }
          : s
      )
    );
  };

  const updateSection = (slideId: string, sectionId: string, updates: Partial<SlideSection>) => {
    setSlides((prev) =>
      prev.map((s) =>
        s.id === slideId
          ? {
              ...s,
              sections: (s.sections || []).map((sec) => (sec.id === sectionId ? { ...sec, ...updates } : sec)),
            }
          : s
      )
    );
  };

  // Generate MLA citations for all images used
  const generateMLACitations = () => {
    const citations: string[] = [];
    slides.forEach((slide, index) => {
      if (slide.imageSource) {
        const source = slide.imageSource;
        let citation = '';
        
        switch (source.type) {
          case 'wikimedia':
            citation = `"${source.title}." Wikimedia Commons${source.author ? `, by ${source.author}` : ''}. Web. ${source.accessDate}. <${source.url}>.`;
            break;
          case 'unsplash':
            citation = `${source.author}. "${source.title}." Unsplash. Web. ${source.accessDate}. <${source.url}>.`;
            break;
          case 'upload':
            citation = `"${source.title}." Personal Collection. ${source.accessDate}.`;
            break;
        }
        
        if (citation) {
          citations.push(`${index + 1}. ${citation}`);
        }
      }
    });
    return citations;
  };

  const deleteSection = (slideId: string, sectionId: string) => {
    setSlides((prev) =>
      prev.map((s) =>
        s.id === slideId ? { ...s, sections: (s.sections || []).filter((sec) => sec.id !== sectionId) } : s
      )
    );
  };

  // Image search functions
  const searchUnsplash = async (query: string) => {
    const key = unsplashKey;
    if (!key) {
      toast({ title: "Clé API requise", description: "Ajoutez votre Unsplash Access Key.", variant: "destructive" });
      return [] as UnsplashImage[];
    }
    const url = `https://api.unsplash.com/search/photos?query=${encodeURIComponent(query)}&per_page=15&client_id=${key}`;
    const res = await fetch(url);
    if (!res.ok) throw new Error("Unsplash error");
    const data = await res.json();
    return (data.results || []) as UnsplashImage[];
  };

  const searchWikimedia = async (query: string) => {
    try {
      // Search for files on Wikimedia Commons with better filtering
      const searchUrl = `https://commons.wikimedia.org/w/api.php?action=query&generator=search&gsrsearch=filetype:image ${encodeURIComponent(
        query
      )}&gsrlimit=20&prop=imageinfo&iiprop=url|extmetadata|size&iiurlwidth=300&format=json&origin=*`;
      
      const res = await fetch(searchUrl);
      if (!res.ok) {
        console.error('Wikimedia API response not ok:', res.status, res.statusText);
        throw new Error(`Wikimedia API error: ${res.status}`);
      }
      
      const data = await res.json();
      console.log('Wikimedia API response:', data);
      
      if (data.error) {
        console.error('Wikimedia API error:', data.error);
        throw new Error(`Wikimedia API error: ${data.error.info}`);
      }
      
      const pages = data?.query?.pages || {};
      const items: WikiImage[] = Object.values(pages)
        .filter((p: any) => {
          // Only include pages that have valid image information
          const hasImageInfo = p?.imageinfo?.[0]?.url;
          const isImage = p?.title?.startsWith('File:') && (
            p.title.toLowerCase().includes('.jpg') ||
            p.title.toLowerCase().includes('.jpeg') ||
            p.title.toLowerCase().includes('.png') ||
            p.title.toLowerCase().includes('.gif') ||
            p.title.toLowerCase().includes('.svg') ||
            p.title.toLowerCase().includes('.webp')
          );
          return hasImageInfo && isImage;
        })
        .map((p: any) => {
          const ii = p.imageinfo[0];
          const meta = ii?.extmetadata || {};
          
          // Extract clean author name
          const authorRaw = meta?.Artist?.value || meta?.Credit?.value || 'Unknown';
          const author = authorRaw.replace(/<[^>]+>/g, '').trim() || 'Unknown';
          
          // Extract license info
          const license = meta?.LicenseShortName?.value || meta?.License?.value || 'CC License';
          
          // Use thumbnail if available, otherwise full image
          const thumb = ii.thumburl || ii.url;
          
          return {
            id: String(p.pageid),
            thumb: thumb,
            full: ii.url,
            title: p.title.replace('File:', '').replace(/\.(jpg|jpeg|png|gif|svg|webp)$/i, ''),
            author: author,
            license: license,
          } as WikiImage;
        });
      
      console.log('Processed Wikimedia results:', items);
      return items;
    } catch (error) {
      console.error('Wikimedia search error:', error);
      throw error;
    }
  };

  const searchImages = async (query: string) => {
    if (!query.trim()) {
      toast({ title: "Recherche requise", description: "Entrez un terme.", variant: "destructive" });
      return;
    }
    setSearching(true);
    setWikiResults([]); // Clear previous results
    setUnsplashResults([]);
    
    try {
      if (provider === "wikimedia") {
        console.log('Searching Wikimedia for:', query);
        const results = await searchWikimedia(query);
        console.log('Got Wikimedia results:', results.length);
        setWikiResults(results);
        
        if (results.length === 0) {
          toast({ 
            title: "Aucun résultat", 
            description: "Aucune image Creative Commons trouvée. Essayez d'autres mots-clés.", 
            variant: "default" 
          });
        } else {
          toast({ 
            title: "Images trouvées", 
            description: `${results.length} images Creative Commons trouvées.`, 
            variant: "default" 
          });
        }
      } else {
        const results = await searchUnsplash(query);
        setUnsplashResults(results);
        
        if (results.length === 0) {
          toast({ 
            title: "Aucun résultat", 
            description: "Aucune image Unsplash trouvée. Essayez d'autres mots-clés.", 
            variant: "default" 
          });
        }
      }
    } catch (error) {
      console.error('Search error:', error);
      const errorMessage = error instanceof Error ? error.message : "Erreur inconnue";
      toast({ 
        title: "Erreur de recherche", 
        description: `${provider === 'wikimedia' ? 'Creative Commons' : 'Unsplash'}: ${errorMessage}`, 
        variant: "destructive" 
      });
    } finally {
      setSearching(false);
    }
  };

  const selectImage = (imageUrl: string, source?: Partial<ImageSource>) => {
    const imageSource: ImageSource = {
      type: source?.type || (provider === 'wikimedia' ? 'wikimedia' : 'unsplash'),
      url: imageUrl,
      title: source?.title || 'Untitled',
      author: source?.author || 'Unknown',
      license: source?.license || 'Unknown',
      accessDate: new Date().toISOString().split('T')[0]
    };
    updateSlide(currentSlide.id, { imageUrl, imageSource });
    setShowImageSearch(false);
  };

  // Extract metadata from Wikimedia Commons URL
  const getWikimediaMetadata = async (imageUrl: string) => {
    try {
      // Extract filename from URL
      const filename = imageUrl.split('/').pop();
      if (!filename) return null;
      
      // Get file info from Wikimedia API
      const apiUrl = `https://commons.wikimedia.org/w/api.php?action=query&titles=File:${encodeURIComponent(filename)}&prop=imageinfo&iiprop=extmetadata&format=json&origin=*`;
      
      const response = await fetch(apiUrl);
      const data = await response.json();
      
      const pages = data?.query?.pages;
      if (!pages) return null;
      
      const pageId = Object.keys(pages)[0];
      const page = pages[pageId];
      
      if (!page?.imageinfo?.[0]?.extmetadata) return null;
      
      const meta = page.imageinfo[0].extmetadata;
      
      return {
        title: filename.replace(/\.(jpg|jpeg|png|gif|svg|webp)$/i, ''),
        author: meta?.Artist?.value?.replace(/<[^>]+>/g, '').trim() || 'Unknown',
        license: meta?.LicenseShortName?.value || meta?.License?.value || 'CC License',
        description: meta?.ImageDescription?.value?.replace(/<[^>]+>/g, '').trim() || ''
      };
    } catch (error) {
      console.error('Error fetching Wikimedia metadata:', error);
      return null;
    }
  };

  // Handle direct image URL input
  const handleDirectImageUrl = async () => {
    if (!directImageUrl.trim()) {
      toast({ title: "URL requise", description: "Entrez une URL d'image.", variant: "destructive" });
      return;
    }
    
    setLoadingDirectImage(true);
    
    try {
      // Check if it's a valid image URL
      const isValidImageUrl = /\.(jpg|jpeg|png|gif|svg|webp)(\?.*)?$/i.test(directImageUrl);
      if (!isValidImageUrl) {
        toast({ title: "URL invalide", description: "L'URL doit pointer vers une image (jpg, png, gif, svg, webp).", variant: "destructive" });
        return;
      }
      
      let imageSource: ImageSource;
      
      // Check if it's a Wikimedia Commons image
      if (directImageUrl.includes('upload.wikimedia.org') || directImageUrl.includes('commons.wikimedia.org')) {
        const metadata = await getWikimediaMetadata(directImageUrl);
        
        imageSource = {
          type: 'wikimedia',
          url: directImageUrl,
          title: metadata?.title || 'Wikimedia Image',
          author: metadata?.author || 'Unknown',
          license: metadata?.license || 'CC License',
          accessDate: new Date().toISOString().split('T')[0]
        };
        
        toast({ 
          title: "Image Wikimedia ajoutée", 
          description: metadata ? `Par ${metadata.author} (${metadata.license})` : "Métadonnées extraites", 
          variant: "default" 
        });
      } else {
        // Generic image URL
        const urlObj = new URL(directImageUrl);
        const filename = urlObj.pathname.split('/').pop() || 'Image';
        
        imageSource = {
          type: 'upload',
          url: directImageUrl,
          title: filename.replace(/\.(jpg|jpeg|png|gif|svg|webp)$/i, ''),
          author: urlObj.hostname,
          license: 'External Source',
          accessDate: new Date().toISOString().split('T')[0]
        };
        
        toast({ 
          title: "Image externe ajoutée", 
          description: `Source: ${urlObj.hostname}`, 
          variant: "default" 
        });
      }
      
      updateSlide(currentSlide.id, { imageUrl: directImageUrl, imageSource });
      setShowImageSearch(false);
      setDirectImageUrl('');
      
    } catch (error) {
      console.error('Error processing direct image URL:', error);
      toast({ title: "Erreur", description: "Impossible de traiter l'URL de l'image.", variant: "destructive" });
    } finally {
      setLoadingDirectImage(false);
    }
  };

  // File upload handling
  const handleFileUpload = (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (file) {
      const reader = new FileReader();
      reader.onload = (e) => {
        const imageUrl = e.target?.result as string;
        const imageSource: ImageSource = {
          type: 'upload',
          url: imageUrl,
          title: file.name,
          author: 'User Upload',
          license: 'Personal Use',
          accessDate: new Date().toISOString().split('T')[0]
        };
        updateSlide(currentSlide.id, { imageUrl, imageSource });
      };
      reader.readAsDataURL(file);
    }
  };

  // Export/Import functions
  const exportSlides = () => {
    const data = { title: presentationTitle, slides };
    const blob = new Blob([JSON.stringify(data, null, 2)], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `${presentationTitle.replace(/[^a-zA-Z0-9]/g, "-")}.json`;
    a.click();
    URL.revokeObjectURL(url);
  };

  const importSlides = (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const data = JSON.parse(e.target?.result as string);
        if (data.slides && Array.isArray(data.slides)) {
          setSlides(data.slides);
          setPresentationTitle(data.title || "Présentation Importée");
          setCurrentSlideIndex(0);
          toast({ title: "Import réussi", description: "Présentation importée avec succès." });
        } else {
          throw new Error("Format invalide");
        }
      } catch (error) {
        toast({ title: "Erreur d'import", description: "Fichier JSON invalide.", variant: "destructive" });
      }
    };
    reader.readAsText(file);
  };

  const exportHTML = () => {
    const fontStacksCSS = Object.entries(FONT_STACKS)
      .map(([key, stack]) => `.font-${key} { font-family: ${stack}; }`)
      .join('\n        ');

    const slidesHTML = slides.map((slide, index) => {

                    >
                      <div className="flex items-center justify-between">
                        <span className="text-sm font-medium">{index + 1}. {slide.title}</span>
                        <div className="flex gap-1">
                          <Button
                            size="sm"
                            variant="ghost"
                            onClick={(e) => {
                              e.stopPropagation();
                              moveSlide(slide.id, 'up');
                            }}
                            disabled={index === 0}
                          >
                            <ArrowUp className="h-3 w-3" />
                          </Button>
                          <Button
                            size="sm"
                            variant="ghost"
                            onClick={(e) => {
                              e.stopPropagation();
                              moveSlide(slide.id, 'down');
                            }}
                            disabled={index === slides.length - 1}
                          >
                            <ArrowDown className="h-3 w-3" />
                          </Button>
                          <Button
                            size="sm"
                            variant="ghost"
                            onClick={(e) => {
                              e.stopPropagation();
                              deleteSlide(slide.id);
                            }}
                          >
                            <Trash2 className="h-3 w-3" />
                          </Button>
                        </div>
                      </div>
                      <div className="text-xs text-gray-500 mt-1">
                        {slide.sections && slide.sections.length > 0 
                          ? `${slide.sections.length} section(s)`
                          : 'Mode legacy'
                        }
                      </div>
                    </div>
                  ))}
                </div>
              </CardContent>
            </Card>
          </div>

          {/* Editor */}
          <div className="lg:col-span-2">
            <Card>
              <CardHeader>
                <CardTitle>Édition - Diapositive {currentSlideIndex + 1}</CardTitle>
              </CardHeader>
              <CardContent className="space-y-6">
                
                {/* Basic slide properties */}
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div>
                    <Label htmlFor="slide-title">Titre</Label>
                    <Input
                      id="slide-title"
                      value={currentSlide?.title || ''}
                      onChange={(e) => updateSlide(currentSlide.id, { title: e.target.value })}
                    />
                  </div>
                  <div>
                    <Label htmlFor="slide-type">Type</Label>
                    <select
                      id="slide-type"
                      value={currentSlide?.type || 'content'}
                      onChange={(e) => updateSlide(currentSlide.id, { type: e.target.value as SlideType })}
                      className="w-full p-2 border border-gray-300 rounded-md"
                    >
                      <option value="title">Titre</option>
                      <option value="content">Contenu</option>
                      <option value="image">Image</option>
                    </select>
                  </div>
                </div>

                {/* Legacy content (backward compatibility) */}
                <div>
                  <Label htmlFor="slide-content">Contenu Legacy (affiché si aucune section)</Label>
                  <Textarea
                    id="slide-content"
                    value={currentSlide?.content || ''}
                    onChange={(e) => updateSlide(currentSlide.id, { content: e.target.value })}
                    rows={3}
                  />
                </div>

                {/* Background and colors */}
                <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                  <div>
                    <Label htmlFor="bg-color">Couleur de fond</Label>
                    <Input
                      id="bg-color"
                      type="color"
                      value={currentSlide?.backgroundColor || '#3B82F6'}
                      onChange={(e) => updateSlide(currentSlide.id, { backgroundColor: e.target.value })}
                    />
                  </div>
                  <div>
                    <Label htmlFor="text-color">Couleur du texte legacy</Label>
                    <Input
                      id="text-color"
                      type="color"
                      value={currentSlide?.textColor || '#ffffff'}
                      onChange={(e) => updateSlide(currentSlide.id, { textColor: e.target.value })}
                    />
                  </div>
                  <div>
                    <Label>Contraste</Label>
                    <div className={`flex items-center gap-2 text-sm ${contrastWarning.warn ? 'text-red-600' : 'text-green-600'}`}>
                      {contrastWarning.warn && <AlertTriangle className="h-4 w-4" />}
                      Ratio: {contrastWarning.ratio.toFixed(1)}
                    </div>
                  </div>
                </div>

                {/* Image settings */}
                <div>
                  <Label>Image de fond</Label>
                  <div className="flex gap-2 mt-2">
                    <Dialog open={showImageSearch} onOpenChange={setShowImageSearch}>
                      <DialogTrigger asChild>
                        <Button variant="outline" size="sm">
                          <Search className="h-4 w-4 mr-2" />
                          Rechercher
                        </Button>
                      </DialogTrigger>
                      <DialogContent className="max-w-4xl max-h-[80vh]">
                        <DialogHeader>
                          <DialogTitle>Ajouter une image</DialogTitle>
                        </DialogHeader>
                        <div className="space-y-6">
                          {/* Direct URL Input */}
                          <div className="space-y-2">
                            <h3 className="text-sm font-medium">URL directe d'image</h3>
                            <div className="flex gap-2">
                              <Input
                                placeholder="https://upload.wikimedia.org/wikipedia/commons/..."
                                value={directImageUrl}
                                onChange={(e) => setDirectImageUrl(e.target.value)}
                                className="flex-1"
                              />
                              <Button 
                                onClick={handleDirectImageUrl} 
                                disabled={loadingDirectImage}
                                size="sm"
                              >
                                {loadingDirectImage ? 'Chargement...' : 'Ajouter'}
                              </Button>
                            </div>
                            <p className="text-xs text-gray-500">
                              Collez une URL d'image directe. Les images Wikimedia Commons auront automatiquement leurs métadonnées extraites pour la citation MLA.
                            </p>
                          </div>
                          
                          <div className="relative">
                            <div className="absolute inset-0 flex items-center">
                              <span className="w-full border-t" />
                            </div>
                            <div className="relative flex justify-center text-xs uppercase">
                              <span className="bg-background px-2 text-muted-foreground">Ou rechercher</span>
                            </div>
                          </div>
                          
                          <div className="space-y-2">
                            <h3 className="text-sm font-medium">Recherche d'images</h3>
                            <div className="flex gap-2">
                              <select
                                value={provider}
                                onChange={(e) => setProvider(e.target.value as Provider)}
                                className="p-2 border rounded"
                              >
                              <option value="wikimedia">Wikimedia Commons (Creative Commons)</option>
                              <option value="unsplash">Unsplash</option>
                            </select>
                            {provider === 'unsplash' && (
                              <Input
                                placeholder="Unsplash Access Key"
                                value={unsplashKey}
                                onChange={(e) => setUnsplashKey(e.target.value)}
                                className="flex-1"
                              />
                            )}
                          </div>
                          <div className="flex gap-2">
                            <Input
                              placeholder="Rechercher des images..."
                              value={searchQuery}
                              onChange={(e) => setSearchQuery(e.target.value)}
                              onKeyPress={(e) => e.key === 'Enter' && searchImages(searchQuery)}
                            />
                            <Button onClick={() => searchImages(searchQuery)} disabled={searching}>
                              <Search className="h-4 w-4" />
                            </Button>
                          </div>
                          <div className="grid grid-cols-3 gap-4 max-h-96 overflow-y-auto">
                            {provider === 'wikimedia' 
                              ? wikiResults.map((img) => (
                                  <div
                                    key={img.id}
                                    className="cursor-pointer border rounded overflow-hidden hover:shadow-lg"
                                    onClick={() => selectImage(img.full, {
                                      type: 'wikimedia',
                                      title: img.title,
                                      author: img.author,
                                      license: img.license,
                                      url: img.full
                                    })}
                                  >
                                    <img src={img.thumb} alt={img.title} className="w-full h-32 object-cover" />
                                    <div className="p-2">
                                      <p className="text-xs text-gray-600 truncate">{img.title}</p>
                                      {img.license && <p className="text-xs text-gray-500">{img.license}</p>}
                                    </div>
                                  </div>
                                ))
                              : unsplashResults.map((img) => (
                                  <div
                                    key={img.id}
                                    className="cursor-pointer border rounded overflow-hidden hover:shadow-lg"
                                    onClick={() => selectImage(img.urls.regular, {
                                      type: 'unsplash',
                                      title: img.alt_description || 'Untitled',
                                      author: img.user.name,
                                      url: img.urls.regular
                                    })}
                                  >
                                    <img src={img.urls.small} alt={img.alt_description} className="w-full h-32 object-cover" />
                                    <div className="p-2">
                                      <p className="text-xs text-gray-600 truncate">{img.alt_description}</p>
                                      <p className="text-xs text-gray-500">by {img.user.name}</p>
                                    </div>
                                  </div>
                                ))
                            }
                          </div>
                        </div>
                      </div>
                      </DialogContent>
                    </Dialog>
                    
                    <Button
                      variant="outline"
                      size="sm"
                      onClick={() => fileInputRef.current?.click()}
                    >
                      <Upload className="h-4 w-4 mr-2" />
                      Téléverser
                    </Button>
                    
                    <input
                      type="file"
                      ref={fileInputRef}
                      onChange={handleFileUpload}
                      accept="image/*"
                      className="hidden"
                    />
                  </div>
                  
                  {currentSlide?.imageUrl && (
                    <div className="mt-2">
                      <img
                        src={currentSlide.imageUrl}
                        alt="Fond"
                        className="w-full h-32 object-cover rounded border"
                      />
                    </div>
                  )}
                </div>

                {/* Overlay settings */}
                <div className="border-t pt-4">
                  <Label className="text-base font-semibold mb-3 block">Options de superposition</Label>
                  <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
                    <div>
                      <Label htmlFor="overlay-type">Type</Label>
                      <select
                        id="overlay-type"
                        className="w-full p-2 border border-gray-300 rounded-md"
                        value={currentSlide?.overlayType || 'none'}
                        onChange={(e) => updateSlide(currentSlide.id, { overlayType: e.target.value as OverlayType })}
                      >
                        <option value="none">Aucune</option>
                        <option value="horizontal">Bande horizontale</option>
                        <option value="sides">Bordures latérales</option>
                      </select>
                    </div>
                    
                    <div>
                      <Label htmlFor="overlay-color">Couleur</Label>
                      <Input
                        id="overlay-color"
                        type="color"
                        value={currentSlide?.overlayColor || '#000000'}
                        onChange={(e) => updateSlide(currentSlide.id, { overlayColor: e.target.value })}
                        disabled={currentSlide?.overlayType === 'none'}
                      />
                    </div>
                    
                    <div>
                      <Label htmlFor="overlay-opacity">Opacité ({Math.round((currentSlide?.overlayOpacity || 0.5) * 100)}%)</Label>
                      <Input
                        id="overlay-opacity"
                        type="range"
                        min="0.1"
                        max="0.9"
                        step="0.1"
                        value={currentSlide?.overlayOpacity || 0.5}
                        onChange={(e) => updateSlide(currentSlide.id, { overlayOpacity: parseFloat(e.target.value) })}
                        disabled={currentSlide?.overlayType === 'none'}
                      />
                    </div>
                    
                    <div>
                      <Label htmlFor="overlay-transition">Transition de l'Overlay</Label>
                      <select
                        id="overlay-transition"
                        value={currentSlide?.overlayTransition || 'none'}
                        onChange={(e) => updateSlide(currentSlide.id, { overlayTransition: e.target.value as TransitionType })}
                        className="w-full p-2 border rounded text-sm"
                        disabled={currentSlide?.overlayType === 'none'}
                      >
                        <option value="none">Aucune</option>
                        <option value="fade">Fondu</option>
                        <option value="slide">Glissement</option>
                        <option value="zoom">Zoom</option>
                        <option value="bounce">Rebond</option>
                        <option value="flip">Rotation</option>
                        <option value="rotate">Rotation 3D</option>
                      </select>
                    </div>
                    
                    <div>
                      <Label htmlFor="text-align">Alignement legacy</Label>
                      <select
                        id="text-align"
                        className="w-full p-2 border border-gray-300 rounded-md"
                        value={currentSlide?.textAlign || 'center'}
                        onChange={(e) => updateSlide(currentSlide.id, { textAlign: e.target.value as TextAlign })}
                      >
                        <option value="left">Gauche</option>
                        <option value="center">Centre</option>
                        <option value="right">Droite</option>
                      </select>
                    </div>
                  </div>
                </div>

                {/* Sections Editor */}
                <div className="border-t pt-4">
                  <div className="flex items-center justify-between mb-4">
                    <Label className="text-base font-semibold">Sections de texte</Label>
                    <Button
                      onClick={() => addSection(currentSlide.id)}
                      size="sm"
                      variant="outline"
                    >
                      <Plus className="h-4 w-4 mr-2" />
                      Ajouter Section
                    </Button>
                  </div>
                  
                  <div className="space-y-4">
                    {(currentSlide?.sections || []).map((section, index) => (
                      <Card key={section.id} className="p-4">
                        <div className="flex items-center justify-between mb-3">
                          <h4 className="font-medium">Section {index + 1}</h4>
                          <Button
                            onClick={() => deleteSection(currentSlide.id, section.id)}
                            size="sm"
                            variant="ghost"
                          >
                            <Trash2 className="h-4 w-4" />
                          </Button>
                        </div>
                        
                        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                          <div>
                            <Label>Titre (optionnel)</Label>
                            <Input
                              value={section.heading || ''}
                              onChange={(e) => updateSection(currentSlide.id, section.id, { heading: e.target.value })}
                              placeholder="Titre de la section..."
                            />
                          </div>
                          <div>
                            <Label>Police</Label>
                            <select
                              value={section.fontFamily}
                              onChange={(e) => updateSection(currentSlide.id, section.id, { fontFamily: e.target.value as FontFamily })}
                              className="w-full p-2 border border-gray-300 rounded-md"
                            >
                              <option value="system">Système</option>
                              <option value="serif">Serif</option>
                              <option value="mono">Monospace</option>
                              <option value="display">Display</option>
                              <option value="hand">Manuscrite</option>
                            </select>
                          </div>
                        </div>
                        
                        <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mt-4">
                          <div>
                            <Label>Taille (rem)</Label>
                            <Input
                              type="number"
                              step="0.1"
                              min="0.5"
                              max="5"
                              value={section.fontSize}
                              onChange={(e) => updateSection(currentSlide.id, section.id, { fontSize: parseFloat(e.target.value) || 1.1 })}
                            />
                          </div>
                          <div>
                            <Label>Couleur</Label>
                            <Input
                              type="color"
                              value={section.color}
                              onChange={(e) => updateSection(currentSlide.id, section.id, { color: e.target.value })}
                            />
                          </div>
                          <div>
                            <Label>Alignement</Label>
                            <select
                              value={section.align}
                              onChange={(e) => updateSection(currentSlide.id, section.id, { align: e.target.value as TextAlign })}
                              className="w-full p-2 border border-gray-300 rounded-md"
                            >
                              <option value="left">Gauche</option>
                              <option value="center">Centre</option>
                              <option value="right">Droite</option>
                            </select>
                          </div>
                        </div>
                        
                        <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mt-4">
                          <div>
                            <Label>Transition</Label>
                            <select
                              value={section.transition || 'none'}
                              onChange={(e) => updateSection(currentSlide.id, section.id, { transition: e.target.value as TransitionType })}
                              className="w-full p-2 border border-gray-300 rounded-md"
                            >
                              <option value="none">Aucune</option>
                              <option value="fade">Fondu</option>
                              <option value="slide">Glissement</option>
                              <option value="zoom">Zoom</option>
                              <option value="bounce">Rebond</option>
                              <option value="flip">Rotation</option>
                              <option value="rotate">Rotation 3D</option>
                            </select>
                          </div>
                          <div>
                            <Label>Délai (secondes)</Label>
                            <Input
                              type="number"
                              step="0.1"
                              min="0"
                              max="5"
                              value={section.transitionDelay || 0}
                              onChange={(e) => updateSection(currentSlide.id, section.id, { transitionDelay: parseFloat(e.target.value) || 0 })}
                            />
                          </div>
                        </div>
                        
                        <div className="mt-4">
                          <Label>Texte</Label>
                          <Textarea
                            value={section.text}
                            onChange={(e) => updateSection(currentSlide.id, section.id, { text: e.target.value })}
                            rows={3}
                            placeholder="Contenu de la section..."
                          />
                        </div>
                        
                        <div className="mt-4 grid grid-cols-1 md:grid-cols-2 gap-4">
                          <XPositionSlider
                            section={section}
                            onUpdate={(updates) => updateSection(currentSlide.id, section.id, updates)}
                          />
                          <YPositionSlider
                            section={section}
                            onUpdate={(updates) => updateSection(currentSlide.id, section.id, updates)}
                          />
                        </div>
                      </Card>
                    ))}
                  </div>
                  
                  {(!currentSlide?.sections || currentSlide.sections.length === 0) && (
                    <div className="text-center py-6 text-gray-500">
                      <Type className="h-12 w-12 mx-auto mb-2 opacity-50" />
                      <p>Aucune section. Le titre/contenu legacy sera affiché.</p>
                      <Button
                        onClick={() => addSection(currentSlide.id)}
                        variant="outline"
                        className="mt-2"
                      >
                        <Plus className="h-4 w-4 mr-2" />
                        Créer la première section
                      </Button>
                    </div>
                  )}
                </div>

              </CardContent>
            </Card>
          </div>

          {/* Preview */}
          <div className="lg:col-span-1">
            <Card>
              <CardHeader>
                <CardTitle>Aperçu</CardTitle>
              </CardHeader>
              <CardContent>
                <div 
                  className="relative w-full aspect-video border rounded-lg overflow-hidden cursor-pointer"
                  style={{
                    background: currentSlide?.imageUrl 
                      ? `linear-gradient(rgba(0,0,0,0.3), rgba(0,0,0,0.3)), url('${currentSlide.imageUrl}')` 
                      : currentSlide?.backgroundColor || '#3B82F6',
                    backgroundSize: 'cover',
                    backgroundPosition: 'center'
                  }}
                  onClick={() => setIsPreviewMode(true)}
                >
                  {/* Overlays */}
                  {currentSlide?.overlayType === 'horizontal' && (
                    <div 
                      className="absolute inset-0"
                      style={{
                        background: hexWithOpacity(currentSlide.overlayColor || '#000000', currentSlide.overlayOpacity || 0.5),
                        top: '30%',
                        bottom: '30%'
                      }}
                    />
                  )}
                  
                  {currentSlide?.overlayType === 'sides' && (
                    <div 
                      className="absolute inset-0"
                      style={{
                        background: `linear-gradient(90deg, 
                          ${hexWithOpacity(currentSlide.overlayColor || '#000000', currentSlide.overlayOpacity || 0.5)} 0%, 
                          transparent 33.33%, 
                          transparent 66.66%, 
                          ${hexWithOpacity(currentSlide.overlayColor || '#000000', currentSlide.overlayOpacity || 0.5)} 100%)`
                      }}
                    />
                  )}
                  
                  {/* Content */}
                  {currentSlide?.sections && currentSlide.sections.length > 0 ? (
                    currentSlide.sections.map(section => renderSection(section))
                  ) : (
                    <div className={`relative z-10 px-4 max-w-sm ${
                      currentSlide?.textAlign === 'left' ? 'text-left' : 
                      currentSlide?.textAlign === 'right' ? 'text-right' : 'text-center'
                    }`}>
                      <h2 className="font-bold mb-4 text-white drop-shadow-lg text-xl">
                        {currentSlide?.title}
                      </h2>
                      <p className="text-sm leading-relaxed text-white drop-shadow whitespace-pre-line">
                        {currentSlide?.content}
                      </p>
                    </div>
                  )}
                </div>
                
                <div className="mt-4 text-center">
                  <Button
                    onClick={() => setIsPreviewMode(true)}
                    className="w-full"
                  >
                    <Eye className="h-4 w-4 mr-2" />
                    Aperçu Plein Écran
                  </Button>
                </div>
              </CardContent>
            </Card>
          </div>

        </div>
      </div>
    </div>
  );
}
